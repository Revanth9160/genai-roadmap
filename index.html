1. Structured Output with Pydantic Schema & Tool Calling 

from pydantic import BaseModel, Field
from openai import OpenAI

class ResearchSummary(BaseModel):
    title: str = Field(description="Main title of the summary")
    key_takeaways: list[str] = Field(description="List of 3 primary insights")
    confidence_score: float = Field(description="Confidence rating between 0 and 1")

client = OpenAI()
response = client.beta.chat.completions.parse(
    model="gpt-4o-mini",
    messages=[{"role": "user", "content": "Summarize recent advances in Agentic AI workflows."}],
    response_format=ResearchSummary,
)
print(response.choices[0].message.parsed)

3. FAISS Vector Search Implementation 

import faiss
import numpy as np

dimension = 1536 # OpenAI embedding dimensions
index = faiss.IndexFlatL2(dimension)

# Add embeddings array
vectors = np.random.random((100, dimension)).astype('float32')
index.add(vectors)

# Query top k nearest neighbors
k = 3
query_vector = np.random.random((1, dimension)).astype('float32')
distances, indices = index.search(query_vector, k)
print("Indices of nearest chunks:", indices)
